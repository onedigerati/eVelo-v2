import { it } from 'vitest'
import { writeFileSync } from 'node:fs'
import marketJson from '/home/jonah/repos/eVelo-v2/src/data/market.json'
import { CITIES } from '/home/jonah/repos/eVelo-v2/src/data/cities'
import { DEFAULT_INPUTS } from '/home/jonah/repos/eVelo-v2/src/engine/defaults'
import { simulate } from '/home/jonah/repos/eVelo-v2/src/engine/simulate'
import { solveSpending } from '/home/jonah/repos/eVelo-v2/src/engine/solve'
import type { MarketData } from '/home/jonah/repos/eVelo-v2/src/engine/types'

const SOLVE_TARGET = 0.9 // = src/state/sim.ts SOLVE_TARGET
it('sanity', () => {
  const market = marketJson as unknown as MarketData
  const d = DEFAULT_INPUTS
  const lines = [
    `DEFAULT_INPUTS: age ${d.age}, portfolio ${d.portfolio}, horizon ${d.horizonYears}y, home ${d.homeCityId}, dest ${d.destCityId}, move-in ${d.moveInYears}y, paths ${d.paths}, seed ${d.seed}; solver target ${SOLVE_TARGET}`,
    'retireIn | savings/yr | success | p50 net worth end | solver maxSpendHome | maxSpendDest',
  ]
  for (const savings of [0, 60000]) for (const n of [0, 2, 4, 5]) {
    const inp = { ...d, retireInYears: n, savingsWhileWorking: savings }
    const r = simulate(market, CITIES, inp)
    const s = solveSpending(market, CITIES, inp, SOLVE_TARGET)
    lines.push(`${n} | ${savings} | ${(r.successRate * 100).toFixed(1)}% | ${Math.round(r.netWorth.p50.at(-1)!)} | ${Math.round(s.maxSpendHome)} | ${s.maxSpendDest == null ? '-' : Math.round(s.maxSpendDest)}`)
  }
  const out = lines.join('\n')
  console.log(out)
  writeFileSync('<scratchpad>/sanity-out.txt', out + '\n')
}, 600000)

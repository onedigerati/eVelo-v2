import { it, expect } from 'vitest'
import { writeFileSync } from 'node:fs'
import marketJson from '../../data/market.json'
import { runAtlas } from '../atlas'
import { DEFAULT_INPUTS } from '../defaults'
import { createRecorder, runPaths } from '../kernel'
import { prepare } from '../prepare'
import { findCity, kernelConfig, makeScenarios, normalizeInputs, resolveDestination, simulate } from '../simulate'
import { solveSpending } from '../solve'
import { CITIES_FIX } from './fixtures'
const market = marketJson as any
it('dump', () => {
  const variant = process.env.VARIANT // 'old' | 'zero' | 'missing'
  const mk = (o: any) => { const x: any = { ...DEFAULT_INPUTS, ...o }; if (variant === 'zero') x.existingLoan = 0; else delete x.existingLoan; return x }
  const cases = [
    {}, { seed: 7, paths: 700 }, { destCityId: CITIES_FIX[CITIES_FIX.length - 1].id, paths: 500 },
    ...CITIES_FIX.slice(0, 6).map((c: any) => ({ destCityId: c.id, paths: 300, seed: 3 })),
    { budget: { ...DEFAULT_INPUTS.budget }, otherIncome: 0, loanRate: 0.09, maxLtv: 0.4, paths: 400 },
    { portfolio: 0, paths: 100 },
  ]
  const out: any = { sims: [], recs: [], solve: null, atlas: null }
  let openingSaleKeys = 0
  for (const c of cases) {
    const inp = mk(c)
    const r: any = simulate(market, CITIES_FIX, inp); delete r.elapsedMs; delete r.inputs
    // openingSale is a field new in this stage; for no balance it must be null, then it is dropped so the rest compares with 75ee416
    if ('openingSale' in r) { expect(r.openingSale).toBeNull(); delete r.openingSale; openingSaleKeys++ }
    out.sims.push(r)
    const inputs = normalizeInputs(inp)
    const prep = prepare(market, inputs.holdings, inputs.outlook)
    const home = findCity(CITIES_FIX, inputs.homeCityId)
    const dest = resolveDestination(inputs, home, findCity(CITIES_FIX, inputs.destCityId))
    const cfg = kernelConfig(inputs, dest)
    const sc = makeScenarios(prep, inputs, inputs.paths)
    const rec: any = createRecorder(sc.paths, Math.floor(cfg.T / 12))
    const clean = runPaths(sc, cfg, rec)
    const o: any = { clean }
    for (const k of Object.keys(rec)) o[k] = ArrayBuffer.isView(rec[k]) ? Array.from(rec[k]) : rec[k]
    out.recs.push(o)
  }
  out.solve = [0.85, 0.9, 0.5].map((t) => solveSpending(market, CITIES_FIX, mk({ paths: 600 }), t))
  out.atlas = runAtlas(market, CITIES_FIX, mk({ paths: 400 }), CITIES_FIX.map((c: any) => c.id))
  writeFileSync(process.env.OUT!, JSON.stringify(out))
  console.log(`variant=${variant} cases=${cases.length} openingSale keys dropped=${openingSaleKeys}`)
})

import type { MarketData, PlanInputs } from '../types'

  { symbol: 'BND', alpha: 0.003, beta: 0.1, noise: 0.01, first: 120 },
])

  console.log('PROBE simulate old fields (no elapsedMs, no yearly):', sha(rp.map((inp) => { const { elapsedMs: _e, yearly: _y, ...rest } = simulate(real, CITIES_FIX, inp) as any; return rest })))
